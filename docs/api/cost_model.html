cost_model

A3 交易成本与滑点模型（固定成本 + 买卖价差 + 冲击成本 + 篮子成本）。

设计目标：

  • 逐笔成本可追溯：每笔成交返回 CostBreakdown，内含逐项 CostDetailItem 明细。
  • 与 engine 解耦：engine 通过 CostModel.compute 接口注入调用，不 import 本模块。
  • 参数集中配置：所有费率来自 CostConfig（cost_config.py），默认值注明"以券商为准需确认"。

成本构成（成交方向不同，符号方向相反）：

  1. 固定成本：佣金（按成交额，有最低佣金）、印花税（仅卖出，ETF 免）、过户费（ETF 免）。
  2. 买卖价差：半价差模型，买入在基准价上加半价差，卖出减半价差。
  3. 冲击成本：按成交量参与率（订单量/成交量）的冲击函数（sqrt 或 linear）。
  4. 篮子成本：逐只计费 + 篮子整体滑点加成。

 45@dataclass
 46class CostBreakdown:
 47    """单笔成交的成本明细分解。"""
 48
 49    symbol: str
 50    side: str
 51    quantity: float
 52    base_price: float  # 基准价（撮合参考价，如开盘价）
 53    exec_price: float  # 价差/冲击调整后的成交价
 54    commission: float
 55    stamp_tax: float
 56    transfer_fee: float
 57    total_fee: float  # 固定费用合计 = 佣金 + 印花税 + 过户费
 58    spread_cost: float  # 半价差成本（金额）
 59    impact_cost: float  # 冲击成本（金额）
 60    slippage_cost: float  # 滑点成本合计 = 价差 + 冲击
 61    gross_value: float  # 基准价口径成交额
 62    net_cash: float  # 成交净现金流（买为负、卖为正，含固定费用）
 63    details: list[CostDetailItem] = field(default_factory=list)
 64
 65    @property
 66    def exec_value(self) -> float:
 67        """成交额（按滑点后成交价）。"""
 68        return self.exec_price * self.quantity
 69
 70    def to_dict(self) -> dict:
 71        """转为 dict（细节列表也展开，便于落盘/审计）。"""
 72        d = asdict(self)
 73        d["details"] = [asdict(x) for x in self.details]
 74        return d

 97class CostModel:
 98    """交易成本模型主类（可注入回测引擎）。"""
 99
100    def __init__(self, config: CostConfig | None = None):
101        self.config = config or CostConfig()
102        self.config.validate()
103
104    # ------------------------------------------------------------------
105    # 单笔成本
106    # ------------------------------------------------------------------
107    def compute(
108        self,
109        side: str,
110        quantity: float,
111        price: float,
112        symbol: str = "",
113        is_etf: bool = False,
114        volume: float = 0.0,
115    ) -> CostBreakdown:
116        """计算单笔成交的成本明细。
117
118        参数：
119        - side: 方向 BUY/SELL
120        - quantity: 成交数量
121        - price: 基准价（撮合参考价，如开盘价）
122        - symbol: 证券代码（仅用于明细标识）
123        - is_etf: 是否 ETF（决定印花税/过户费豁免）
124        - volume: 该 bar 成交量（用于计算冲击参与率，0 则无冲击）
125        """
126        side = normalize_side(side)
127        cfg = self.config
128        qty = float(quantity)
129        base = float(price)
130
131        # 半价差与冲击（以"率"表示，相对基准价）
132        half_spread_rate = cfg.spread_rate / 2.0
133        impact_rate = self._impact_rate(qty, volume)
134
135        slip_rate = half_spread_rate + impact_rate
136        exec_price = base * (1.0 + slip_rate) if side == BUY else base * (1.0 - slip_rate)
137
138        exec_value = exec_price * qty
139
140        # 固定费用
141        commission = self._commission(exec_value)
142        stamp_tax = (
143            0.0
144            if (is_etf and cfg.etf_exempt_stamp_tax) or side == BUY
145            else exec_value * cfg.stamp_tax_rate
146        )
147        transfer_fee = (
148            0.0 if (is_etf and cfg.etf_exempt_transfer_fee) else exec_value * cfg.transfer_fee_rate
149        )
150        total_fee = commission + stamp_tax + transfer_fee
151
152        # 滑点金额（相对基准价的口径）
153        spread_cost = base * half_spread_rate * qty
154        impact_cost = base * impact_rate * qty
155        slippage_cost = spread_cost + impact_cost
156
157        gross_value = base * qty
158        net_cash = -(exec_value + total_fee) if side == BUY else (exec_value - total_fee)
159
160        details = [
161            CostDetailItem(
162                "佣金",
163                commission,
164                f"费率 {cfg.commission_rate:.4%}，最低 {cfg.min_commission:.2f} 元",
165            ),
166            CostDetailItem(
167                "印花税",
168                stamp_tax,
169                f"卖出 {cfg.stamp_tax_rate:.4%}"
170                + ("（ETF 免）" if is_etf and cfg.etf_exempt_stamp_tax else ""),
171            ),
172            CostDetailItem(
173                "过户费",
174                transfer_fee,
175                f"双向 {cfg.transfer_fee_rate:.4%}"
176                + ("（ETF 免）" if is_etf and cfg.etf_exempt_transfer_fee else ""),
177            ),
178            CostDetailItem("价差成本(半价差)", spread_cost, f"半价差 {half_spread_rate:.4%}"),
179            CostDetailItem(
180                "冲击成本",
181                impact_cost,
182                f"{cfg.impact_model} 模型，参与率 {self._participation(qty, volume):.4%}",
183            ),
184        ]
185
186        return CostBreakdown(
187            symbol=symbol,
188            side=side,
189            quantity=qty,
190            base_price=base,
191            exec_price=exec_price,
192            commission=commission,
193            stamp_tax=stamp_tax,
194            transfer_fee=transfer_fee,
195            total_fee=total_fee,
196            spread_cost=spread_cost,
197            impact_cost=impact_cost,
198            slippage_cost=slippage_cost,
199            gross_value=gross_value,
200            net_cash=net_cash,
201            details=details,
202        )
203
204    # ------------------------------------------------------------------
205    # 篮子成本（逐只计费 + 篮子整体滑点加成）
206    # ------------------------------------------------------------------
207    def compute_basket(self, legs: list[dict]) -> BasketCost:
208        """计算 ETF 篮子同步执行的整体成本。
209
210        legs 每项为 dict，键同 compute 参数：
211            {symbol, side, quantity, price, is_etf, volume}
212        返回逐只 CostBreakdown + 篮子整体滑点加成。
213        """
214        if not legs:
215            return BasketCost()
216
217        breakdowns = [self.compute(**leg) for leg in legs]
218        gross = sum(b.gross_value for b in breakdowns)
219        # 篮子整体滑点加成：对整体成交额按 bps 计
220        basket_slippage = gross * self.config.basket_slippage_bps / 10000.0
221        total_fee = sum(b.total_fee for b in breakdowns)
222        total_slippage = sum(b.slippage_cost for b in breakdowns) + basket_slippage
223
224        return BasketCost(
225            legs=breakdowns,
226            basket_slippage=basket_slippage,
227            total_fee=total_fee,
228            total_slippage=total_slippage,
229            total_gross_value=gross,
230        )
231
232    # ------------------------------------------------------------------
233    # 内部工具
234    # ------------------------------------------------------------------
235    def _commission(self, exec_value: float) -> float:
236        """佣金（含最低佣金约束）。"""
237        cfg = self.config
238        if exec_value <= 0:
239            return 0.0
240        return max(exec_value * cfg.commission_rate, cfg.min_commission)
241
242    def _participation(self, quantity: float, volume: float) -> float:
243        """成交参与率 = 订单量/成交量，封顶于 participation_cap。"""
244        if volume is None or volume <= 0:
245            return 0.0
246        return min(float(quantity) / float(volume), self.config.participation_cap)
247
248    def _impact_rate(self, quantity: float, volume: float) -> float:
249        """冲击成本率（相对基准价的占比）。"""
250        cfg = self.config
251        participation = self._participation(quantity, volume)
252        if participation <= 0:
253            return 0.0
254        if cfg.impact_model == IMPACT_SQRT:
255            return cfg.impact_coef * np.sqrt(participation)
256        if cfg.impact_model == IMPACT_LINEAR:
257            return cfg.impact_coef * participation
258        raise ValueError(f"未知冲击模型: {cfg.impact_model}")

261def sensitivity_table(
262    base: CostConfig,
263    varied_param: str,
264    values: list[float],
265    side: str = BUY,
266    quantity: float = 10000,
267    price: float = 3.0,
268    volume: float = 1_000_000,
269    is_etf: bool = True,
270) -> list[dict]:
271    """成本敏感性分析：固定其他参数，仅变动一个参数，输出总成本变化。
272
273    返回 [{param_value, commission, stamp_tax, transfer_fee, spread_cost,
274            impact_cost, slippage_cost, total_fee, total_cost}]，
275    其中 total_cost = 固定费用 + 滑点成本（买方口径的总成本）。
276    用于输出"成本敏感"结论：参数对总成本的敏感度。
277    """
278    rows = []
279    for v in values:
280        cfg = CostConfig(**{**base.__dict__, varied_param: v})
281        model = CostModel(cfg)
282        b = model.compute(
283            side=side, quantity=quantity, price=price, symbol="510300", is_etf=is_etf, volume=volume
284        )
285        total_cost = b.total_fee + b.slippage_cost
286        rows.append(
287            {
288                varied_param: v,
289                "commission": b.commission,
290                "stamp_tax": b.stamp_tax,
291                "transfer_fee": b.transfer_fee,
292                "spread_cost": b.spread_cost,
293                "impact_cost": b.impact_cost,
294                "slippage_cost": b.slippage_cost,
295                "total_fee": b.total_fee,
296                "total_cost": total_cost,
297            }
298        )
299    return rows

251        cfg = self.config

253        if participation <= 0:
254            return 0.0

256            return cfg.impact_coef * np.sqrt(participation)
257        if cfg.impact_model == IMPACT_LINEAR:
