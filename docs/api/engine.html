1945def _no_init_or_replace_init(self, *args, **kwargs):
1946    cls = type(self)
1947
1948    if cls._is_protocol:
1949        raise TypeError('Protocols cannot be instantiated')
1950
1951    # Already using a custom `__init__`. No need to calculate correct
1952    # `__init__` to call. This can lead to RecursionError. See bpo-45121.
1953    if cls.__init__ is not _no_init_or_replace_init:
1954        return
1955
1956    # Initially, `__init__` of a protocol subclass is set to `_no_init_or_replace_init`.
1957    # The first instantiation of the subclass will call `_no_init_or_replace_init` which
1958    # searches for a proper new `__init__` in the MRO. The new `__init__`
1959    # replaces the subclass' old `__init__` (ie `_no_init_or_replace_init`). Subsequent
1960    # instantiation of the protocol subclass will thus use the new
1961    # `__init__` and no longer call `_no_init_or_replace_init`.
1962    for base in cls.__mro__:
1963        init = base.__dict__.get('__init__', _no_init_or_replace_init)
1964        if init is not _no_init_or_replace_init:
1965            cls.__init__ = init
1966            break
1967    else:
1968        # should not happen
1969        cls.__init__ = object.__init__
1970
1971    cls.__init__(self, *args, **kwargs)

class EventEngine: 
 83class EventEngine:
 84    """事件驱动回测主引擎。"""
 85
 86    def __init__(
 87        self,
 88        config: EngineConfig | None = None,
 89        data: dict[str, pd.DataFrame] | pd.DataFrame | None = None,
 90        cost_model: CostModelLike | None = None,
 91        strategy: Strategy | None = None,
 92    ):
 93        self.config = config or EngineConfig()
 94        self.config.validate()
 95
 96        self.data = self._normalize_data({} if data is None else data)
 97        self.cost_model = cost_model
 98        self.matching = MatchingEngine(cost_model) if cost_model is not None else None
 99        self.strategy = strategy
100
101        self.rng = np.random.default_rng(self.config.seed)
102        self._callbacks: dict[str, list[Callable]] = defaultdict(list)
103        self._order_seq = 0
104        self.logger = self._setup_logger()
105
106        # 预计算 master 时间轴与各标的 next-bar 映射
107        self._master_index: pd.DatetimeIndex = self._build_master_index()
108        self._tsets: dict[str, set] = {sym: set(df.index) for sym, df in self.data.items()}
109        self._next_map: dict[str, dict[pd.Timestamp, pd.Timestamp]] = {}
110        for sym, df in self.data.items():
111            idx = df.index
112            self._next_map[sym] = dict(itertools.pairwise(idx))
113
114    # ------------------------------------------------------------------
115    # 插件/回调注册
116    # ------------------------------------------------------------------
117    def register_strategy(self, strategy: Strategy) -> None:
118        """挂载策略。"""
119        self.strategy = strategy
120
121    def register_callback(self, event_cls: type, fn: Callable) -> None:
122        """按事件类型注册回调（如 FillEvent、PortfolioEvent、MarketDataEvent）。"""
123        self._callbacks[event_cls.__name__].append(fn)
124
125    def _emit(self, event: Event) -> None:
126        """把事件分发给对应类型的回调。"""
127        for fn in self._callbacks.get(type(event).__name__, []):
128            fn(event, self)
129
130    # ------------------------------------------------------------------
131    # 运行
132    # ------------------------------------------------------------------
133    def run(self) -> EngineResult:
134        """执行事件循环，返回 EngineResult。"""
135        if self.matching is None:
136            raise EngineError("引擎未注入成本模型，无法撮合")
137
138        portfolio = Portfolio(self.config.initial_cash)
139        accounting = Accounting()
140        pending: list[Order] = []
141        orders_log: list[Order] = []
142        self._order_seq = 0
143        last_prices: dict[str, float] = {}
144
145        for t in self._master_index:
146            # 1) 构造当前 bar 的行情事件
147            bars_t: dict[str, MarketDataEvent] = {}
148            for sym, df in self.data.items():
149                if t in self._tsets[sym]:
150                    row = df.loc[t]
151                    bars_t[sym] = self._make_market_event(t, sym, row)
152                    last_prices[sym] = bars_t[sym].close
153            if not bars_t:
154                continue
155            self._emit_market_events(bars_t)
156
157            # 2) 撮合到期订单（next_bar 单在下一 bar 开盘成交）
158            remaining: list[Order] = []
159            for order in pending:
160                if order.fill_timestamp == t:
161                    self._execute(order, bars_t, portfolio, accounting)
162                else:
163                    remaining.append(order)
164            pending = remaining
165
166            # 3) 策略生成信号 → 订单
167            if self.strategy is not None:
168                signals = self.strategy.on_bar(t, bars_t, portfolio)
169                for sig in signals:
170                    order = self._create_order(sig, t)
171                    orders_log.append(order)
172                    self._emit(
173                        OrderEvent(
174                            timestamp=order.timestamp,
175                            order_id=order.order_id,
176                            symbol=order.symbol,
177                            side=order.side.value,
178                            quantity=order.quantity,
179                            order_type=order.order_type.value,
180                            limit_price=order.limit_price,
181                            status=order.status.value,
182                            fill_timestamp=order.fill_timestamp,
183                        )
184                    )
185                    if self.config.fill_mode == FILL_OPEN:
186                        # 当 bar 开盘价撮合（注意：可能与信号使用当 bar 收盘价产生前视）
187                        self._execute(order, bars_t, portfolio, accounting)
188                    else:  # next_bar
189                        nt = self._next_map.get(order.symbol, {}).get(t)
190                        if nt is None:
191                            order.status = OrderStatus.REJECTED
192                            self.logger.debug("订单 %s 无下一 bar，拒绝", order.order_id)
193                        else:
194                            order.status = OrderStatus.PENDING
195                            order.fill_timestamp = nt
196                            pending.append(order)
197
198            # 4) 期末盯市（用各标的最近已知收盘价）
199            prices = dict(last_prices)
200            rec = accounting.mark_to_market(t, portfolio, prices)
201            self._emit(
202                PortfolioEvent(
203                    timestamp=t,
204                    cash=rec["cash"],
205                    positions=rec["positions"],
206                    market_value=rec["market_value"],
207                    equity=rec["equity"],
208                    available_cash=rec["available_cash"],
209                    total_fees=rec["total_fees"],
210                )
211            )
212
213        return EngineResult(
214            equity_curve=accounting.equity_curve(),
215            fills=accounting.fills_frame(),
216            orders=pd.DataFrame([o.to_dict() for o in orders_log]),
217            portfolio=portfolio,
218            config=self.config,
219        )
220
221    # ------------------------------------------------------------------
222    # 内部：订单执行
223    # ------------------------------------------------------------------
224    def _execute(
225        self,
226        order: Order,
227        bars_t: dict[str, MarketDataEvent],
228        portfolio: Portfolio,
229        accounting: Accounting,
230    ) -> None:
231        """撮合单笔订单并应用成交（含资金校验）。"""
232        bar = bars_t.get(order.symbol)
233        if bar is None:
234            order.status = OrderStatus.CANCELLED
235            self.logger.debug("订单 %s 在计划时点无行情，取消", order.order_id)
236            return
237
238        fill = self.matching.match(order, bar, is_etf=self._is_etf(order.symbol))
239        if fill is None:
240            order.status = OrderStatus.REJECTED
241            self.logger.debug("订单 %s 限价未成交，拒绝", order.order_id)
242            return
243
244        if order.side == OrderSide.BUY:
245            cash_out = fill.exec_price * fill.quantity + fill.total_fee
246            if not portfolio.can_afford(cash_out):
247                order.status = OrderStatus.REJECTED
248                self.logger.debug("订单 %s 资金不足（需 %.2f），拒绝", order.order_id, cash_out)
249                return
250
251        order.status = OrderStatus.FILLED
252        portfolio.apply_fill(fill)
253        accounting.record_fill(fill)
254        self._emit(fill)
255
256    def _create_order(self, sig: SignalEvent, t: pd.Timestamp) -> Order:
257        """由信号构造订单对象。"""
258        self._order_seq += 1
259        order_id = f"ORD-{self._order_seq:06d}"
260        side = coerce_side(sig.side)
261        quantity = float(sig.quantity)
262        if not self.config.allow_fractional:
263            quantity = float(int(quantity))
264        is_limit = (sig.order_type == "limit") or (sig.limit_price is not None)
265        order_type = OrderType.LIMIT if is_limit else OrderType.MARKET
266        return Order(
267            order_id=order_id,
268            symbol=sig.symbol,
269            side=side,
270            quantity=quantity,
271            order_type=order_type,
272            limit_price=sig.limit_price,
273            timestamp=t,
274            status=OrderStatus.CREATED,
275        )
276
277    def _is_etf(self, symbol: str) -> bool:
278        if self.config.etf_symbols is not None:
279            return symbol in set(self.config.etf_symbols)
280        return is_etf_symbol(symbol)

282    # ------------------------------------------------------------------
283    # 内部：数据与时间轴
284    # ------------------------------------------------------------------
285    def _normalize_data(self, data) -> dict[str, pd.DataFrame]:
286        """把输入数据归一化为 {symbol: DataFrame}（宽表 dict 或含 symbol 列的长表）。"""
287        if isinstance(data, dict):
288            return {sym: self._prepare_symbol_df(sym, df) for sym, df in data.items()}
289        if isinstance(data, pd.DataFrame) and sc.COL_SYMBOL in data.columns:
290            out = {}
291            for sym, sub in data.groupby(sc.COL_SYMBOL):

135        if self.matching is None:
136            raise EngineError("引擎未注入成本模型，无法撮合")

141        orders_log: list[Order] = []

143        last_prices: dict[str, float] = {}
144
145        for t in self._master_index:
146            # 1) 构造当前 bar 的行情事件
147            bars_t: dict[str, MarketDataEvent] = {}
148            for sym, df in self.data.items():
149                if t in self._tsets[sym]:

151                    bars_t[sym] = self._make_market_event(t, sym, row)
152                    last_prices[sym] = bars_t[sym].close
153            if not bars_t:
154                continue
155            self._emit_market_events(bars_t)
156
157            # 2) 撮合到期订单（next_bar 单在下一 bar 开盘成交）
158            remaining: list[Order] = []
159            for order in pending:
160                if order.fill_timestamp == t:
161                    self._execute(order, bars_t, portfolio, accounting)
162                else:
163                    remaining.append(order)
164            pending = remaining
165
166            # 3) 策略生成信号 → 订单
167            if self.strategy is not None:
168                signals = self.strategy.on_bar(t, bars_t, portfolio)
169                for sig in signals:

171                    orders_log.append(order)
172                    self._emit(
173                        OrderEvent(

175                            order_id=order.order_id,

193                        else:

196                            pending.append(order)
197
198            # 4) 期末盯市（用各标的最近已知收盘价）
