etf.etf_signal

B2 折溢价信号层：滚动 z-score/分位数 + 开平仓阈值 + 强平 + 方向判定。

防前视约束（硬约束）：

  • 所有滚动统计（z-score / 分位数）只在窗口 [t-window+1, t] 内计算，绝不使用 t 之后的数据（rolling 天然因果，无全样本标准化）。
  • 开仓阈值 = 单位交易成本 + 缓冲（成本来自 A3 cost_model），逻辑上覆盖成本。

方向约定：

  • 溢价（premium > 0）：ETF 二级价高于净值 → 卖 ETF、买篮子（direction=short）。
  • 折价（premium < 0）：ETF 二级价低于净值 → 买 ETF、卖篮子（direction=long）。

输出：开/平仓信号事件流 DataFrame（字段含 ts/action/direction/threshold_basis）， 并附 signal_to_engine_events 把一行信号转换为 A2 引擎的 SignalEvent 列表。

 58def rolling_zscore(premium: pd.Series, window: int) -> pd.Series:
 59    """滚动 z-score：z_t = (x_t - mean(x_{t-w+1..t})) / std(x_{t-w+1..t})。
 60
 61    只在滚动窗口内计算（无前视）；窗口内标准差为 0（常数序列）时 z 置 NaN。
 62    """
 63    x = pd.to_numeric(premium, errors="coerce")
 64    mu = x.rolling(window).mean()
 65    sd = x.rolling(window).std()
 66    return (x - mu) / sd.replace(0, np.nan)
 67
 68
 69def rolling_quantile(premium: pd.Series, window: int) -> pd.Series:
 70    """滚动分位数：当前值在窗口 [t-window+1, t] 内的百分位（0~1，无前视）。"""
 71    x = pd.to_numeric(premium, errors="coerce")
 72
 73    def _rank(w: np.ndarray) -> float:
 74        return float((w <= w[-1]).mean())
 75
 76    return x.rolling(window).apply(_rank, raw=True)

 79def entry_threshold(config: ThresholdConfig, unit_cost_rate: float) -> float:
 80    """开仓阈值 = 单位成本 + 缓冲（逻辑上覆盖成本）。"""
 81    return float(unit_cost_rate) + float(config.entry_buffer)

 84def round_trip_cost_rate(
 85    cost_model: CostModel,
 86    price: float,
 87    quantity: float,
 88    is_etf: bool = True,
 89    volume: float = 0.0,
 90) -> float:
 91    """单位（双边往返）成本率 = (一次买 + 一次卖的总成本) / 双边成交额。
 92
 93    成本 = 固定费用（佣金/印花税/过户费）+ 滑点（价差 + 冲击），与 A3 口径一致。
 94    返回比例（如 0.0006 = 6bp），用于开仓阈值"覆盖成本"。
 95    """
 96    buy = cost_model.compute("BUY", quantity, price, is_etf=is_etf, volume=volume)
 97    sell = cost_model.compute("SELL", quantity, price, is_etf=is_etf, volume=volume)
 98    notional = 2.0 * float(price) * float(quantity)
 99    cost = (buy.total_fee + buy.slippage_cost) + (sell.total_fee + sell.slippage_cost)
100    return float(cost / notional) if notional > 0 else 0.0

103class ETFSignalGenerator:
104    """折溢价信号生成器：滚动判据 + 开/平/强平状态机。"""
105
106    def __init__(
107        self,
108        config: ThresholdConfig | None = None,
109        cost_model: CostModel | None = None,
110        unit_cost_rate: float | None = None,
111    ):
112        self.config = config or ThresholdConfig()
113        self.config.validate()
114        if unit_cost_rate is None:
115            model = cost_model or CostModel()
116            unit_cost_rate = round_trip_cost_rate(model, price=3.0, quantity=10000, is_etf=True)
117        self.unit_cost_rate = float(unit_cost_rate)
118        self.threshold = entry_threshold(self.config, self.unit_cost_rate)
119
120    # ------------------------------------------------------------------
121    # 主入口：生成信号事件流
122    # ------------------------------------------------------------------
123    def generate(self, premium: pd.Series) -> pd.DataFrame:
124        """由折溢价序列生成开/平仓信号事件流。
125
126        返回 DataFrame，仅含 open/close 事件行，字段：
127        ts / action / direction / etf_side / premium / zscore / quantile /
128        entry_threshold / threshold_basis / stop_loss_level。
129        """
130        cfg = self.config
131        x = pd.to_numeric(premium, errors="coerce")
132        z = rolling_zscore(x, cfg.zscore_window)
133        q = rolling_quantile(x, cfg.zscore_window)
134
135        events: list[dict] = []
136        position: dict | None = None  # {direction, entry_ts, entry_premium}
137        force_close_min = cfg.force_close_minute_of_day()
138        min_hold = pd.Timedelta(minutes=cfg.min_holding_minutes)
139
140        for ts, prem, zs, qt in zip(x.index, x.values, z.values, q.values, strict=False):
141            if not np.isfinite(prem) or not np.isfinite(zs) or not np.isfinite(qt):
142                continue  # 数据不足（warmup / 缺失），不产生信号
143
144            force_close_now = (ts.hour * 60 + ts.minute) >= force_close_min
145
146            if position is None:
147                # 收盘强平只作用于持仓；空仓时收盘前不再新开仓（避免隔夜敞口）
148                if force_close_now:
149                    continue
150                direction, basis = self._entry_signal(prem, zs, qt)
151                if direction is not None:
152                    position = {
153                        "direction": direction,
154                        "entry_ts": ts,
155                        "entry_premium": prem,
156                    }
157                    events.append(
158                        self._make_event(
159                            ts, ACTION_OPEN, direction, prem, zs, qt, basis, position=None
160                        )
161                    )
162            else:
163                direction = position["direction"]
164                stop_level = self._stop_level(position)
165                hit_stop = (direction == DIR_LONG and prem <= stop_level) or (
166                    direction == DIR_SHORT and prem >= stop_level
167                )
168                reverted = (direction == DIR_LONG and zs >= cfg.zscore_exit) or (
169                    direction == DIR_SHORT and zs <= -cfg.zscore_exit
170                )
171                holding_ok = (ts - position["entry_ts"]) >= min_hold
172
173                if holding_ok and force_close_now:
174                    reason = f"平仓:收盘强平(>{cfg.force_close_time})"
175                elif holding_ok and hit_stop:
176                    reason = f"平仓:止损(突破{cfg.stop_loss_bp}bp)"
177                elif holding_ok and reverted:
178                    reason = "平仓:均值回归/归零"
179                else:
180                    reason = None
181
182                if reason is not None:
183                    events.append(
184                        self._make_event(
185                            ts, ACTION_CLOSE, direction, prem, zs, qt, reason, position=position
186                        )
187                    )
188                    position = None
189
190        out = pd.DataFrame(events, columns=_EVENT_COLUMNS)
191        if out.empty:
192            return out
193        out[COL_TS] = pd.to_datetime(out[COL_TS])
194        return out.sort_values(COL_TS).reset_index(drop=True)
195
196    # ------------------------------------------------------------------
197    # 内部：开仓 / 止损 / 事件行
198    # ------------------------------------------------------------------
199    def _entry_signal(self, prem: float, zs: float, qt: float):
200        """开仓判据：|溢价| 覆盖成本 + z-score/分位数极端。返回 (direction, basis)。"""
201        cfg = self.config
202        if prem >= self.threshold:
203            if cfg.use_quantile:
204                cond, desc = qt >= cfg.quantile_entry, f"分位数>={cfg.quantile_entry:.2f}"
205            else:
206                cond, desc = zs >= cfg.zscore_entry, f"z>={cfg.zscore_entry:.2f}"
207            if cond:
208                basis = (
209                    f"开仓:溢价{prem:.5f}>=阈值{self.threshold:.5f}"
210                    f"(成本{self.unit_cost_rate:.5f}+缓冲{cfg.entry_buffer:.5f}) 且 {desc}"
211                )
212                return DIR_SHORT, basis
213        elif prem <= -self.threshold:
214            if cfg.use_quantile:
215                cond, desc = (
216                    qt <= 1.0 - cfg.quantile_entry,
217                    f"分位数<={1.0 - cfg.quantile_entry:.2f}",
218                )
219            else:
220                cond, desc = zs <= -cfg.zscore_entry, f"z<={-cfg.zscore_entry:.2f}"
221            if cond:
222                basis = (
223                    f"开仓:折价{prem:.5f}<=-阈值{-self.threshold:.5f}"
224                    f"(成本{self.unit_cost_rate:.5f}+缓冲{cfg.entry_buffer:.5f}) 且 {desc}"
225                )
226                return DIR_LONG, basis
227        return None, None
228
229    def _stop_level(self, position: dict) -> float:
230        """持仓止损线（溢价朝不利方向移动 stop_loss_bp 即触发）。"""
231        stop_bp = self.config.stop_loss_bp / 1e4
232        if position["direction"] == DIR_LONG:
233            return position["entry_premium"] - stop_bp
234        return position["entry_premium"] + stop_bp
235
236    def _make_event(
237        self,
238        ts: pd.Timestamp,
239        action: str,
240        direction: str,
241        prem: float,
242        zs: float,
243        qt: float,
244        reason: str,
245        position: dict | None,
246    ) -> dict:
247        etf_side = SELL if direction == DIR_SHORT else BUY
248        return {
249            COL_TS: ts,
250            COL_ACTION: action,
251            COL_DIRECTION: direction,
252            COL_ETF_SIDE: etf_side,
253            COL_PREMIUM: float(prem),
254            COL_ZSCORE: float(zs),
255            COL_QUANTILE: float(qt),
256            COL_ENTRY_THRESHOLD: self.threshold,
257            COL_REASON: reason,
258            COL_STOP_LEVEL: self._stop_level(position) if position is not None else np.nan,
259        }

262def signal_to_engine_events(
263    row,
264    etf_symbol: str,
265    quantity: float,
266    basket: dict[str, float] | None = None,
267) -> list[SignalEvent]:
268    """把一行信号事件转换为 A2 引擎的 ``SignalEvent`` 列表（可被 EventEngine 消费）。
269
270    - 开仓 long（折价）：买 ETF（BUY），卖篮子成分（SELL）。
271    - 开仓 short（溢价）：卖 ETF（SELL），买篮子成分（BUY）。
272    - 平仓：与开仓方向相反。
273    - basket：{symbol: quantity} 篮子成分股目标数量；缺省只输出 ETF 腿。
274    """
275    ts = pd.Timestamp(row[COL_TS])
276    action = str(row[COL_ACTION])
277    direction = str(row[COL_DIRECTION])
278    reason = str(row[COL_REASON])
279
280    if action == ACTION_OPEN:
281        etf_side = SELL if direction == DIR_SHORT else BUY
282        basket_side = BUY if direction == DIR_SHORT else SELL
283    else:  # close：与开仓相反
284        etf_side = BUY if direction == DIR_SHORT else SELL
285        basket_side = SELL if direction == DIR_SHORT else BUY
286
287    events = [
288        SignalEvent(
289            timestamp=ts,
290            symbol=etf_symbol,
291            side=etf_side,
292            quantity=float(quantity),
293            order_type="market",
294            reason=reason,
295        )
296    ]
297    if basket:
298        for sym, qty in basket.items():
299            events.append(
300                SignalEvent(
301                    timestamp=ts,
302                    symbol=str(sym),
303                    side=basket_side,
304                    quantity=float(qty),
305                    order_type="market",
306                    reason=reason,
307                )
308            )
309    return events

134        q = rolling_quantile(x, cfg.zscore_window)
135
136        events: list[dict] = []
137        position: dict | None = None  # {direction, entry_ts, entry_premium}
138        force_close_min = cfg.force_close_minute_of_day()
139        min_hold = pd.Timedelta(minutes=cfg.min_holding_minutes)
140
141        for ts, prem, zs, qt in zip(x.index, x.values, z.values, q.values, strict=False):
142            if not np.isfinite(prem) or not np.isfinite(zs) or not np.isfinite(qt):
143                continue  # 数据不足（warmup / 缺失），不产生信号
144
145            force_close_now = (ts.hour * 60 + ts.minute) >= force_close_min
146
147            if position is None:
148                # 收盘强平只作用于持仓；空仓时收盘前不再新开仓（避免隔夜敞口）
149                if force_close_now:
150                    continue
151                direction, basis = self._entry_signal(prem, zs, qt)
152                if direction is not None:
153                    position = {
154                        "direction": direction,
155                        "entry_ts": ts,
156                        "entry_premium": prem,
157                    }
158                    events.append(
159                        self._make_event(
160                            ts, ACTION_OPEN, direction, prem, zs, qt, basis, position=None
161                        )
162                    )
163            else:
164                direction = position["direction"]
165                stop_level = self._stop_level(position)
166                hit_stop = (direction == DIR_LONG and prem <= stop_level) or (
167                    direction == DIR_SHORT and prem >= stop_level
168                )
169                reverted = (direction == DIR_LONG and zs >= cfg.zscore_exit) or (
170                    direction == DIR_SHORT and zs <= -cfg.zscore_exit
171                )
172                holding_ok = (ts - position["entry_ts"]) >= min_hold
173
174                if holding_ok and force_close_now:
175                    reason = f"平仓:收盘强平(>{cfg.force_close_time})"
176                elif holding_ok and hit_stop:
177                    reason = f"平仓:止损(突破{cfg.stop_loss_bp}bp)"
178                elif holding_ok and reverted:
179                    reason = "平仓:均值回归/归零"
180                else:
181                    reason = None
182
183                if reason is not None:
184                    events.append(
185                        self._make_event(
186                            ts, ACTION_CLOSE, direction, prem, zs, qt, reason, position=position
187                        )
188                    )
189                    position = None
190
191        out = pd.DataFrame(events, columns=_EVENT_COLUMNS)
192        if out.empty:
193            return out
194        out[COL_TS] = pd.to_datetime(out[COL_TS])
195        return out.sort_values(COL_TS).reset_index(drop=True)

由折溢价序列生成开/平仓信号事件流。

返回 DataFrame，仅含 open/close 事件行，字段： ts / action / direction / etf_side / premium / zscore / quantile / entry_threshold / threshold_basis / stop_loss_level。

def rolling_quantile( premium: pandas.core.series.Series, window: int) -> pandas.core.series.Series: 
70def rolling_quantile(premium: pd.Series, window: int) -> pd.Series:
71    """滚动分位数：当前值在窗口 [t-window+1, t] 内的百分位（0~1，无前视）。"""
72    x = pd.to_numeric(premium, errors="coerce")
73
74    def _rank(w: np.ndarray) -> float:
75        return float((w <= w[-1]).mean())
76
77    return x.rolling(window).apply(_rank, raw=True)

def rolling_zscore( premium: pandas.core.series.Series, window: int) -> pandas.core.series.Series: 
59def rolling_zscore(premium: pd.Series, window: int) -> pd.Series:
60    """滚动 z-score：z_t = (x_t - mean(x_{t-w+1..t})) / std(x_{t-w+1..t})。
61
62    只在滚动窗口内计算（无前视）；窗口内标准差为 0（常数序列）时 z 置 NaN。
63    """
64    x = pd.to_numeric(premium, errors="coerce")
65    mu = x.rolling(window).mean()
66    sd = x.rolling(window).std()
67    return (x - mu) / sd.replace(0, np.nan)

滚动 z-score：z_t = (x_t - mean(x_{t-w+1..t})) / std(x_{t-w+1..t})。

def round_trip_cost_rate( cost_model: cost_model.CostModel, price: float, quantity: float, is_etf: bool = True, volume: float = 0.0) -> float: 
 85def round_trip_cost_rate(
 86    cost_model: CostModel,
 87    price: float,
 88    quantity: float,
 89    is_etf: bool = True,
 90    volume: float = 0.0,
 91) -> float:
 92    """单位（双边往返）成本率 = (一次买 + 一次卖的总成本) / 双边成交额。
 93
 94    成本 = 固定费用（佣金/印花税/过户费）+ 滑点（价差 + 冲击），与 A3 口径一致。
 95    返回比例（如 0.0006 = 6bp），用于开仓阈值"覆盖成本"。
 96    """
 97    buy = cost_model.compute("BUY", quantity, price, is_etf=is_etf, volume=volume)
 98    sell = cost_model.compute("SELL", quantity, price, is_etf=is_etf, volume=volume)
 99    notional = 2.0 * float(price) * float(quantity)
100    cost = (buy.total_fee + buy.slippage_cost) + (sell.total_fee + sell.slippage_cost)
101    return float(cost / notional) if notional > 0 else 0.0

283        basket_side = BUY if direction == DIR_SHORT else SELL
284    else:  # close：与开仓相反
