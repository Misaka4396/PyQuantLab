etf.basket_execution

B3 篮子同步执行与申赎模拟。

ETF 套利最大风险是"篮子同步成交困难"。本模块模拟真实执行不确定性：

  • 逐笔时延（可配 1-5 分钟）与部分成交（概率/比例可配）。
  • 滑点：复用 A3 cost_model.compute_basket 逐只计费 + 篮子整体滑点。
  • 敞口：未成交部分产生 tracking error（执行缺口加权标准差）。
  • 申赎（可选，开关默认关闭）：AP 实物/现金申赎、最小申赎单位、费用、T 日确认。 关闭时走二级市场成交路径（compute_basket）；开启时额外输出申赎结果。

执行成本与 A3 一致：本模块不自行发明费率，全部经 CostModel 计费。

 62class ExecutionResult:
 63    """一次篮子执行的完整结果。"""

 94def _weighted_std(values: Sequence[float], weights: Sequence[float]) -> float:
 95    """加权标准差（执行缺口口径）。"""
 96    v = np.asarray(values, dtype=float)
 97    w = np.asarray(weights, dtype=float)
 98    if v.size == 0 or w.sum() <= 0:
 99        return 0.0
100    mean = float(np.average(v, weights=w))
101    var = float(np.average((v - mean) ** 2, weights=w))
102    return float(np.sqrt(var))

105def make_basket_legs(
106    basket: PCFBasket,
107    etf_quantity: float,
108    etf_price: float,
109    stock_prices: dict[str, float],
110    direction: str = "long",
111) -> list[dict]:
112    """由 PCF 篮子构造套利腿（供 BasketExecutor.execute 使用）。
113
114    direction="long"（折价套利）：买 ETF、卖成分股。
115    direction="short"（溢价套利）：卖 ETF、买成分股。
116    成分股数量 = 篮子名义 × 权重 / 价格。
117    """
118    if direction == "long":
119        etf_side, stock_side = BUY, SELL
120    elif direction == "short":
121        etf_side, stock_side = SELL, BUY
122    else:
123        raise ValueError(f"direction 需为 long/short，得到 {direction!r}")
124
125    legs = [{"symbol": basket.etf_code, "side": etf_side, "quantity": float(etf_quantity)}]
126    weights = basket.weights()
127    basket_notional = float(etf_quantity) * float(etf_price)
128    for c in basket.constituents:
129        w = weights.get(c.symbol, 0.0)
130        px = float(stock_prices.get(c.symbol, np.nan)) if c.symbol in stock_prices else np.nan
131        if not np.isfinite(px) or px <= 0 or w <= 0:
132            continue
133        legs.append({"symbol": c.symbol, "side": stock_side, "quantity": basket_notional * w / px})
134    return legs

137class BasketExecutor:
138    """篮子同步执行模拟器。"""
139
140    def __init__(
141        self,
142        config: ExecutionConfig | None = None,
143        cost_model: CostModel | None = None,
144        seed: int | None = None,
145    ):
146        self.config = config or ExecutionConfig()
147        self.config.validate()
148        self.cost_model = cost_model or CostModel()
149        self._rng = np.random.default_rng(seed if seed is not None else self.config.seed)
150
151    # ------------------------------------------------------------------
152    def execute(
153        self,
154        legs: Sequence[dict],
155        prices: dict[str, float],
156        volumes: dict[str, float] | None = None,
157        basket: PCFBasket | None = None,
158        etf_symbol: str | None = None,
159    ) -> ExecutionResult:
160        """模拟一次篮子同步执行。
161
162        - legs：[{symbol, side, quantity}]（ETF 腿 + 成分股腿）。
163        - prices：{symbol: 基准价}。
164        - volumes：{symbol: 成交量}（冲击成本参与率用，缺省 0 → 无冲击）。
165        - basket：PCF 篮子（启用申赎时必需）。
166        - etf_symbol：ETF 腿代码（用于 is_etf 判断与申赎定位）。
167        """
168        cfg = self.config
169        volumes = volumes or {}
170
171        # 1) 逐笔时延 + 部分成交
172        leg_execs: list[LegExecution] = []
173        for leg in legs:
174            sym = str(leg["symbol"])
175            side = str(leg["side"]).upper()
176            qty = float(leg["quantity"])
177            price = float(prices.get(sym, np.nan))
178            delay = float(self._rng.uniform(cfg.delay_minutes_min, cfg.delay_minutes_max))
179            is_partial = bool(self._rng.random() < cfg.partial_fill_prob)
180            fill_ratio = 1.0
181            if is_partial:
182                fill_ratio = float(
183                    self._rng.uniform(cfg.partial_fill_ratio_min, cfg.partial_fill_ratio_max)
184                )
185            filled = qty * fill_ratio
186            leg_execs.append(
187                LegExecution(
188                    symbol=sym,
189                    side=side,
190                    quantity=qty,
191                    filled_quantity=filled,
192                    unfilled_quantity=qty - filled,
193                    ref_price=price,
194                    exec_price=price,
195                    delay_minutes=delay,
196                    is_partial=is_partial,
197                )
198            )
199
200        # 2) 申赎（可选）：开启且有篮子时输出 AP 申赎结果
201        cr = None
202        if cfg.enable_creation_redemption and basket is not None:
203            cr = self._creation_redemption(leg_execs, basket, etf_symbol)
204
205        # 3) 二级市场成本（A3 compute_basket，只对已成交数量计费）
206        etf_code = etf_symbol or (basket.etf_code if basket is not None else "")
207        cost_legs = [
208            {
209                "symbol": le.symbol,
210                "side": le.side,
211                "quantity": le.filled_quantity,
212                "price": le.ref_price,
213                "is_etf": le.symbol == etf_code,
214                "volume": float(volumes.get(le.symbol, 0.0)),
215            }
216            for le in leg_execs
217            if le.filled_quantity > 0
218        ]
219        basket_cost = self.cost_model.compute_basket(cost_legs) if cost_legs else None
220
221        # 回填成交价 / 费用到各腿
222        if basket_cost is not None:
223            by_sym = {b.symbol: b for b in basket_cost.legs}
224            for le in leg_execs:
225                b = by_sym.get(le.symbol)
226                if b is not None:
227                    le.exec_price = b.exec_price
228                    le.total_fee = b.total_fee
229                    le.slippage_cost = b.slippage_cost
230
231        # 4) 敞口 / 跟踪误差
232        total_target = sum(
233            abs(le.ref_price * le.quantity) for le in leg_execs if np.isfinite(le.ref_price)
234        )
235        total_filled = sum(
236            abs(le.ref_price * le.filled_quantity) for le in leg_execs if np.isfinite(le.ref_price)
237        )
238        total_unfilled = sum(
239            abs(le.ref_price * le.unfilled_quantity)
240            for le in leg_execs
241            if np.isfinite(le.ref_price)
242        )
243        exposure = total_unfilled / total_target if total_target > 0 else 0.0
244
245        gaps, weights = [], []
246        for le in leg_execs:
247            if le.quantity > 0 and np.isfinite(le.ref_price):
248                gaps.append(le.unfilled_quantity / le.quantity)
249                weights.append(abs(le.ref_price * le.quantity))
250        tracking_error = _weighted_std(gaps, weights)
251
252        return ExecutionResult(
253            legs=leg_execs,
254            basket_cost=basket_cost,
255            creation_redemption=cr,
256            total_target_value=total_target,
257            total_filled_value=total_filled,
258            total_unfilled_value=total_unfilled,
259            exposure_ratio=exposure,
260            tracking_error=tracking_error,
261        )
262
263    # ------------------------------------------------------------------
264    def _creation_redemption(
265        self,
266        leg_execs: list[LegExecution],
267        basket: PCFBasket,
268        etf_symbol: str | None,
269    ) -> dict | None:
270        """AP 实物/现金申赎模拟（ETF 腿）。"""
271        cfg = self.config
272        etf_leg = None
273        for le in leg_execs:
274            if le.symbol == (etf_symbol or "") or le.symbol == basket.etf_code:
275                etf_leg = le
276                break
277        if etf_leg is None:
278            return None
279
280        units = int(etf_leg.quantity // cfg.creation_unit) if cfg.creation_unit > 0 else 0
281        if units <= 0:
282            return {
283                "type": "insufficient_units",
284                "units": 0,
285                "etf_quantity": 0.0,
286                "fee": 0.0,
287                "cash_component": 0.0,
288                "confirm_day": cfg.confirm_day,
289            }
290        typ = "redeem" if etf_leg.side == SELL else "create"
291        cash_component = basket.cash_component * units
292        fee = cfg.creation_fee + cash_component * cfg.cash_substitute_fee_bp / 10000.0
293        return {
294            "type": typ,
295            "units": units,
296            "etf_quantity": units * cfg.creation_unit,
297            "fee": fee,
298            "cash_component": cash_component,
299            "confirm_day": cfg.confirm_day,
300        }

193                    unfilled_quantity=qty - filled,
194                    ref_price=price,
195                    exec_price=price,
196                    delay_minutes=delay,
197                    is_partial=is_partial,
198                )
199            )
200
201        # 2) 申赎（可选）：开启且有篮子时输出 AP 申赎结果
202        cr = None
203        if cfg.enable_creation_redemption and basket is not None:
204            cr = self._creation_redemption(leg_execs, basket, etf_symbol)
205
206        # 3) 二级市场成本（A3 compute_basket，只对已成交数量计费）
207        etf_code = etf_symbol or (basket.etf_code if basket is not None else "")
208        cost_legs = [
209            {
210                "symbol": le.symbol,
211                "side": le.side,
212                "quantity": le.filled_quantity,
213                "price": le.ref_price,
214                "is_etf": le.symbol == etf_code,
215                "volume": float(volumes.get(le.symbol, 0.0)),
216            }
217            for le in leg_execs
218            if le.filled_quantity > 0
219        ]
220        basket_cost = self.cost_model.compute_basket(cost_legs) if cost_legs else None
221
222        # 回填成交价 / 费用到各腿
223        if basket_cost is not None:
224            by_sym = {b.symbol: b for b in basket_cost.legs}
225            for le in leg_execs:
226                b = by_sym.get(le.symbol)
227                if b is not None:
228                    le.exec_price = b.exec_price
229                    le.total_fee = b.total_fee
230                    le.slippage_cost = b.slippage_cost
231
232        # 4) 敞口 / 跟踪误差
233        total_target = sum(
234            abs(le.ref_price * le.quantity) for le in leg_execs if np.isfinite(le.ref_price)
235        )
236        total_filled = sum(
237            abs(le.ref_price * le.filled_quantity) for le in leg_execs if np.isfinite(le.ref_price)
238        )
239        total_unfilled = sum(
240            abs(le.ref_price * le.unfilled_quantity)
241            for le in leg_execs
242            if np.isfinite(le.ref_price)
243        )
244        exposure = total_unfilled / total_target if total_target > 0 else 0.0
245
246        gaps, weights = [], []
247        for le in leg_execs:
248            if le.quantity > 0 and np.isfinite(le.ref_price):
249                gaps.append(le.unfilled_quantity / le.quantity)
250                weights.append(abs(le.ref_price * le.quantity))
251        tracking_error = _weighted_std(gaps, weights)
252
253        return ExecutionResult(
254            legs=leg_execs,
255            basket_cost=basket_cost,
256            creation_redemption=cr,
257            total_target_value=total_target,
258            total_filled_value=total_filled,
259            total_unfilled_value=total_unfilled,
260            exposure_ratio=exposure,
261            tracking_error=tracking_error,
262        )

def make_basket_legs( basket: etf.pcf_parser.PCFBasket, etf_quantity: float, etf_price: float, stock_prices: dict[str, float], direction: str = 'long') -> list[dict]: 
106def make_basket_legs(
107    basket: PCFBasket,
108    etf_quantity: float,
109    etf_price: float,
110    stock_prices: dict[str, float],
111    direction: str = "long",
112) -> list[dict]:
113    """由 PCF 篮子构造套利腿（供 BasketExecutor.execute 使用）。
114
115    direction="long"（折价套利）：买 ETF、卖成分股。
116    direction="short"（溢价套利）：卖 ETF、买成分股。
117    成分股数量 = 篮子名义 × 权重 / 价格。
118    """
119    if direction == "long":
120        etf_side, stock_side = BUY, SELL
121    elif direction == "short":
122        etf_side, stock_side = SELL, BUY
123    else:
124        raise ValueError(f"direction 需为 long/short，得到 {direction!r}")
125
126    legs = [{"symbol": basket.etf_code, "side": etf_side, "quantity": float(etf_quantity)}]
127    weights = basket.weights()
128    basket_notional = float(etf_quantity) * float(etf_price)
129    for c in basket.constituents:
130        w = weights.get(c.symbol, 0.0)
131        px = float(stock_prices.get(c.symbol, np.nan)) if c.symbol in stock_prices else np.nan
132        if not np.isfinite(px) or px <= 0 or w <= 0:
133            continue
134        legs.append({"symbol": c.symbol, "side": stock_side, "quantity": basket_notional * w / px})
135    return legs
