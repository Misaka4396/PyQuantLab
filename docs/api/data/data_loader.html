统一数据加载入口：PIT 对齐 + 清洗 + 增量更新 + 各 parquet 表读写。

存储目录结构（data_root 默认 ./data_cache/pit）： ohlcv/{symbol}.parquet 行情（长表，index=datetime） adj_factor/{symbol}.parquet 复权因子 constituents/{index_code}.parquet 成分股会员记录 pcf/{etf_code}_{trade_date}.parquet PCF 篮子文件 nav/{etf_code}.parquet NAV/IOPV

设计约定：

  • 所有行情数据落盘时都带 as_of 时间戳（point-in-time）。
  • 落盘的是"原始已归一化"数据；清洗（缺失/异常/停牌/涨跌停）在读取时进行， 这样质量报告仍能基于原始数据统计缺失率与异常率。

 48def detect_limit_status(df: pd.DataFrame, limit_pct: float = 0.10) -> pd.Series:
 49    """按上一交易日收盘价标记涨跌停状态（近似：真实涨跌停价会四舍五入到分）。"""
 50    close = df[sc.COL_CLOSE]
 51    prev_close = close.shift(1)
 52    status = pd.Series(sc.LIMIT_NORMAL, index=df.index, dtype=object)
 53    with np.errstate(invalid="ignore"):
 54        up = prev_close * (1.0 + limit_pct)
 55        down = prev_close * (1.0 - limit_pct)
 56        is_up = (close >= up - 1e-9).fillna(False)
 57        is_down = (close <= down + 1e-9).fillna(False)
 58    status[is_up] = sc.LIMIT_UP
 59    status[is_down] = sc.LIMIT_DOWN
 60    return status

 63def clean_ohlcv(
 64    df: pd.DataFrame, limit_pct: float = 0.10, ffill_close: bool = True
 65) -> pd.DataFrame:
 66    """清洗 OHLCV 长表：去重排序、异常价、缺失值、停牌、涨跌停。
 67
 68    处理步骤：
 69    1. 时间戳统一为 tz-naive，去重（keep=last）并排序；
 70    2. 非正价格（<=0）置为 NaN；
 71    3. 收盘价前向填充（停牌日沿用最近有效价），仍缺失则删除该行；
 72    4. open/high/low 缺失时用收盘价兜底；保证 high>=max(open,close)、low<=min(open,close)；
 73    5. volume/amount 缺失补 0；
 74    6. 标记停牌（volume==0）与涨跌停状态。
 75    """
 76    df = df.copy()
 77    if sc.COL_DATETIME in df.columns:
 78        df = df.set_index(sc.COL_DATETIME)
 79    if not isinstance(df.index, pd.DatetimeIndex):
 80        df.index = pd.to_datetime(df.index)
 81    df.index = normalize_timestamps(df.index)
 82    df = df[~df.index.duplicated(keep="last")]
 83    df = df.sort_index()
 84
 85    for c in sc.OHLCV_REQUIRED:
 86        if c not in df.columns:
 87            df[c] = np.nan
 88        df[c] = pd.to_numeric(df[c], errors="coerce")
 89    # 非正价格视为异常，置 NaN
 90    for c in sc.OHLCV_REQUIRED:
 91        df.loc[df[c] <= 0, c] = np.nan
 92
 93    if ffill_close:
 94        df[sc.COL_CLOSE] = df[sc.COL_CLOSE].ffill()
 95    df = df.dropna(subset=[sc.COL_CLOSE])
 96
 97    for c in (sc.COL_OPEN, sc.COL_HIGH, sc.COL_LOW):
 98        df[c] = df[c].fillna(df[sc.COL_CLOSE])
 99    df[sc.COL_HIGH] = df[[sc.COL_HIGH, sc.COL_OPEN, sc.COL_CLOSE]].max(axis=1)
100    df[sc.COL_LOW] = df[[sc.COL_LOW, sc.COL_OPEN, sc.COL_CLOSE]].min(axis=1)
101
102    if sc.COL_VOLUME not in df.columns:
103        df[sc.COL_VOLUME] = 0.0
104    df[sc.COL_VOLUME] = pd.to_numeric(df[sc.COL_VOLUME], errors="coerce").fillna(0.0)
105    if sc.COL_AMOUNT not in df.columns:
106        df[sc.COL_AMOUNT] = 0.0
107    df[sc.COL_AMOUNT] = pd.to_numeric(df[sc.COL_AMOUNT], errors="coerce").fillna(0.0)
108
109    df[sc.COL_IS_SUSPENDED] = df[sc.COL_VOLUME] <= 0
110    df[sc.COL_LIMIT_STATUS] = detect_limit_status(df, limit_pct)
111    df.loc[df[sc.COL_IS_SUSPENDED], sc.COL_LIMIT_STATUS] = sc.SUSPENDED
112    return df

115def pivot_field(df: pd.DataFrame, field: str) -> pd.DataFrame:
116    """把长表透视成宽表（行=datetime，列=symbol），供回测/组合模块使用。"""
117    if field not in df.columns:
118        raise DataError(f"透视字段不存在: {field}")
119    return df.pivot(columns=sc.COL_SYMBOL, values=field)

159    def _ensure_as_of(self, df: pd.DataFrame, default_lag: str = "0D") -> pd.DataFrame:
160        """缺省时自动生成 as_of = 索引 + lag（数据须带 as_of 时间戳）。"""
161        df = df.copy()
162        if sc.COL_AS_OF not in df.columns:
163            idx = df.index if isinstance(df.index, pd.DatetimeIndex) else pd.to_datetime(df.index)
164            df[sc.COL_AS_OF] = idx + pd.Timedelta(default_lag)
165        return df
166
167    def _prepare_ohlcv(
168        self, symbol: str, df: pd.DataFrame, default_as_of_lag: str = "0D"
169    ) -> pd.DataFrame:
170        """归一化 OHLCV（不落盘）：时间索引、必填列校验、补可选列、去重排序。"""
171        df = df.copy()
172        if sc.COL_DATETIME in df.columns:
173            df = df.set_index(sc.COL_DATETIME)
174        if not isinstance(df.index, pd.DatetimeIndex):
175            df.index = pd.to_datetime(df.index)
176        df.index = normalize_timestamps(df.index)
177        df.index.name = sc.COL_DATETIME
178
179        for c in sc.OHLCV_REQUIRED:
180            if c not in df.columns:
181                raise DataError(f"OHLCV 缺少必需列: {c}")
182        for c in sc.OHLCV_OPTIONAL:
183            if c not in df.columns:
184                df[c] = np.nan
185
186        df = self._ensure_as_of(df, default_as_of_lag)
187        df[sc.COL_SYMBOL] = symbol
188        df = df[sc.OHLCV_STORED_COLUMNS]
189        df = df[~df.index.duplicated(keep="last")].sort_index()
190        return df
191
192    def save_ohlcv(
193        self, symbol: str, df: pd.DataFrame, default_as_of_lag: str = "0D"
194    ) -> pd.DataFrame:
195        """落盘 OHLCV（覆盖写）。返回归一化后的 DataFrame。"""
196        prepared = self._prepare_ohlcv(symbol, df, default_as_of_lag)
197        prepared.to_parquet(self._ohlcv_path(symbol), index=True)
198        return prepared

200    def load_ohlcv(
201        self,
202        symbol: str,
203        start: DateLike | None = None,
204        end: DateLike | None = None,
205        as_of: DateLike | None = None,
206        clean: bool = True,
207    ) -> pd.DataFrame:
208        """读取 OHLCV，支持时间区间与 as_of 过滤，按需清洗。
209
210        - as_of：只返回 ``as_of <= as_of`` 的行（防未来函数）。
211        - clean=True：应用 clean_ohlcv（补停牌/涨跌停标记）。
212        """
213        path = self._ohlcv_path(symbol)
214        if not path.exists():
215            return self._empty_ohlcv()
216        df = pd.read_parquet(path)
217        if start is not None:
218            df = df[df.index >= pd.Timestamp(start)]
219        if end is not None:
220            df = df[df.index <= pd.Timestamp(end)]
221        if as_of is not None:
222            df = slice_as_of(df, as_of)
223        if clean and not df.empty:
224            df = clean_ohlcv(df)
225        return df
226
227    def load_ohlcv_many(
228        self,
229        symbols: Iterable[str],
230        start: DateLike | None = None,
231        end: DateLike | None = None,
232        as_of: DateLike | None = None,
233        clean: bool = True,
234    ) -> pd.DataFrame:
235        """批量读取多只证券，纵向拼接为长表。"""
236        frames = [
237            self.load_ohlcv(s, start=start, end=end, as_of=as_of, clean=clean) for s in symbols
238        ]
239        frames = [f for f in frames if not f.empty]
240        if not frames:
241            return self._empty_ohlcv()
242        return pd.concat(frames).sort_index()

244    def incremental_update_ohlcv(
245        self,
246        symbol: str,
247        new_df: pd.DataFrame,
248        default_as_of_lag: str = "0D",
249    ) -> pd.DataFrame:
250        """增量更新：合并新 bar，按时间戳去重（保留最新），覆盖写回。"""
251        prepared = self._prepare_ohlcv(symbol, new_df, default_as_of_lag)
252        existing = self.load_ohlcv(symbol, clean=False)
253        combined = prepared if existing.empty else pd.concat([existing, prepared])
254        combined = combined[~combined.index.duplicated(keep="last")].sort_index()
255        combined.to_parquet(self._ohlcv_path(symbol), index=True)
256        return combined

291    def save_constituents(self, index_code: str, df: pd.DataFrame) -> pd.DataFrame:
292        """落盘成分股会员记录（幸存者偏差数据）。"""
293        df = df.copy()
294        for c in (sc.COL_SYMBOL, sc.COL_ENTRY_DATE):
295            if c not in df.columns:
296                raise DataError(f"成分股记录缺少必需列: {c}")
297        if sc.COL_EXIT_DATE not in df.columns:
298            df[sc.COL_EXIT_DATE] = pd.NaT
299        if sc.COL_AS_OF not in df.columns:
300            df[sc.COL_AS_OF] = df[sc.COL_ENTRY_DATE]
301        if sc.COL_REASON not in df.columns:
302            df[sc.COL_REASON] = sc.REASON_ADD
303        df[sc.COL_INDEX_CODE] = index_code
304        df[sc.COL_ENTRY_DATE] = pd.to_datetime(df[sc.COL_ENTRY_DATE])
305        df[sc.COL_EXIT_DATE] = pd.to_datetime(df[sc.COL_EXIT_DATE])
306        df[sc.COL_AS_OF] = pd.to_datetime(df[sc.COL_AS_OF])
307        out = df[sc.CONSTITUENTS_STORED_COLUMNS].reset_index(drop=True)
308        out.to_parquet(self._constituents_path(index_code), index=False)
309        return out

327    def save_pcf(self, etf_code: str, trade_date: str, df: pd.DataFrame) -> pd.DataFrame:
328        """落盘 PCF 篮子文件（本批次仅预留 schema，抓取在 B1 完成）。"""

334        df[sc.COL_TRADE_DATE] = pd.Timestamp(trade_date)

338        if sc.COL_AS_OF not in df.columns:

340        out = df[sc.PCF_STORED_COLUMNS].reset_index(drop=True)
341        out.to_parquet(self._pcf_path(etf_code, trade_date), index=False)
342        return out

346    ) -> pd.DataFrame:
347        """读取 PCF 篮子文件；trade_date 为空时返回该 ETF 全部篮子文件。"""

352                return pd.DataFrame(columns=sc.PCF_STORED_COLUMNS)
353            df = pd.read_parquet(path)
354        else:
355            paths = sorted(d.glob(f"{etf_code}_*.parquet"))
356            if not paths:
357                return pd.DataFrame(columns=sc.PCF_STORED_COLUMNS)
358            df = pd.concat([pd.read_parquet(p) for p in paths], ignore_index=True)
359        if as_of is not None:
360            df = slice_as_of(df, as_of)
361        return df

 67    """清洗 OHLCV 长表：去重排序、异常价、缺失值、停牌、涨跌停。
 68
 69    处理步骤：
 70    1. 时间戳统一为 tz-naive，去重（keep=last）并排序；
 71    2. 非正价格（<=0）置为 NaN；
 72    3. 收盘价前向填充（停牌日沿用最近有效价），仍缺失则删除该行；
 73    4. open/high/low 缺失时用收盘价兜底；保证 high>=max(open,close)、low<=min(open,close)；
 74    5. volume/amount 缺失补 0；
 75    6. 标记停牌（volume==0）与涨跌停状态。
 76    """
 77    df = df.copy()
 78    if sc.COL_DATETIME in df.columns:
 79        df = df.set_index(sc.COL_DATETIME)
 80    if not isinstance(df.index, pd.DatetimeIndex):
 81        df.index = pd.to_datetime(df.index)
 82    df.index = normalize_timestamps(df.index)
 83    df = df[~df.index.duplicated(keep="last")]
 84    df = df.sort_index()
 85
 86    for c in sc.OHLCV_REQUIRED:
 87        if c not in df.columns:
 88            df[c] = np.nan
 89        df[c] = pd.to_numeric(df[c], errors="coerce")
 90    # 非正价格视为异常，置 NaN
 91    for c in sc.OHLCV_REQUIRED:
 92        df.loc[df[c] <= 0, c] = np.nan
 93
 94    if ffill_close:
 95        df[sc.COL_CLOSE] = df[sc.COL_CLOSE].ffill()
 96    df = df.dropna(subset=[sc.COL_CLOSE])
 97
 98    for c in (sc.COL_OPEN, sc.COL_HIGH, sc.COL_LOW):
 99        df[c] = df[c].fillna(df[sc.COL_CLOSE])
100    df[sc.COL_HIGH] = df[[sc.COL_HIGH, sc.COL_OPEN, sc.COL_CLOSE]].max(axis=1)
101    df[sc.COL_LOW] = df[[sc.COL_LOW, sc.COL_OPEN, sc.COL_CLOSE]].min(axis=1)
102
103    if sc.COL_VOLUME not in df.columns:
104        df[sc.COL_VOLUME] = 0.0
105    df[sc.COL_VOLUME] = pd.to_numeric(df[sc.COL_VOLUME], errors="coerce").fillna(0.0)
106    if sc.COL_AMOUNT not in df.columns:
107        df[sc.COL_AMOUNT] = 0.0
108    df[sc.COL_AMOUNT] = pd.to_numeric(df[sc.COL_AMOUNT], errors="coerce").fillna(0.0)
109
110    df[sc.COL_IS_SUSPENDED] = df[sc.COL_VOLUME] <= 0
111    df[sc.COL_LIMIT_STATUS] = detect_limit_status(df, limit_pct)
112    df.loc[df[sc.COL_IS_SUSPENDED], sc.COL_LIMIT_STATUS] = sc.SUSPENDED
113    return df
