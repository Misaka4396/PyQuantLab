backtest.metrics_enhanced

A4 绩效评估层（增强版指标，供 B4/C4 联动）。

与旧 backtest/metrics.py（向量化引擎用）解耦，本模块直接消费 A2 事件驱动 引擎的输出：engine/accounting.py 的权益曲线 DataFrame 与 fills_frame() 成交明细 DataFrame。

指标公式（全部在此注明定义，方便与手算对照 / 审计）：

收益

  • 累计收益率 cumulative_return = equity[-1] / equity[0] - 1
  • 年化收益率 annualized_return = (1 + total_return) ** (periods_per_year / n_periods) - 1 其中 n_periods = 收益率观测数（len(returns)，首根 bar 无收益已丢弃）
  • 月度收益 monthly_returns：把日收益率按自然月复利 (1+r).prod()-1
  • 年度收益 yearly_returns：把日收益率按自然年复利

风险

  • 年化波动率 annualized_volatility = std(returns, ddof=1) * sqrt(periods_per_year)
  • 最大回撤 max_drawdown = min(equity / equity.cummax() - 1)
  • Sharpe = mean(excess) / std(excess, ddof=1) * sqrt(periods_per_year)， excess = returns - rf / periods_per_year（rf 为年化无风险利率）
  • Sortino = mean(excess) / std(excess[excess < 0], ddof=1) * sqrt(periods_per_year)
  • Calmar = annualized_return / |max_drawdown|
  • VaR(α) = 收益升序排列后的第 floor((1-α)*n) 个（经验分位数，取更保守一侧）
  • CVaR(α) = 收益 <= VaR 的尾部收益的均值

交易（由成交明细 FIFO 配对得到逐笔平仓交易）

  • 胜率 win_rate = 盈利笔数 / 平仓总笔数
  • 盈亏比 profit_factor = 盈利总额 / |亏损总额|
  • 平均盈亏比 avg_win_loss_ratio = 平均盈利 / |平均亏损|
  • 换手率 turnover = 期间总成交额 / 期间平均权益（双边口径）
  • 平均持仓周期 avg_holding_bars = 平仓交易的平均持仓 bar 数
  • 成本占比 cost_ratio = 累计费用 / 累计成交额

与 C4 联动：compare_is_oos 接收 IS/OOS 两组权益序列做对比（过拟合检测的输入）。

 52# 工具：从 A2 权益曲线 / 成交明细抽取标准输入
 53# ---------------------------------------------------------------------------
 54def _equity_series(equity_curve: pd.DataFrame) -> pd.Series:
 55    """从权益曲线 DataFrame 抽取 equity 列（缺失则报错）。"""
 56    if "equity" not in equity_curve.columns:
 57        raise ValueError("equity_curve 必须包含 equity 列")
 58    s = pd.to_numeric(equity_curve["equity"], errors="coerce")
 59    return s

 62def _returns_from_equity(equity: pd.Series) -> pd.Series:
 63    """由权益序列计算收益率：r_t = equity_t / equity_{t-1} - 1，首元素 NaN 丢弃。"""
 64    r = equity.pct_change(fill_method=None)
 65    return r.dropna()

 68# ---------------------------------------------------------------------------
 69# FIFO 平仓配对：把 A2 成交明细还原为逐笔交易（已实现盈亏）
 70# ---------------------------------------------------------------------------
 71@dataclass
 72class RoundTrip:
 73    """一笔平仓交易（FIFO 配对后）。"""
 74
 75    symbol: str
 76    entry_time: pd.Timestamp
 77    exit_time: pd.Timestamp
 78    quantity: float
 79    entry_price: float
 80    exit_price: float
 81    pnl: float  # 已扣双边费用的净盈亏
 82    pnl_pct: float  # pnl / 建仓成本
 83    win: bool
 84    holding_bars: int
 85
 86
 87def build_round_trips(
 88    fills: pd.DataFrame,
 89    equity_index: pd.DatetimeIndex | None = None,
 90) -> list[RoundTrip]:
 91    """按 symbol 分组、FIFO 配对平仓，返回逐笔交易列表。
 92
 93    费用分摊：买入单位成本 = exec_price + 买入费/买入量；卖出单位收入 = exec_price - 卖出费/卖出量。
 94    未平仓（只有买入没有卖出）的部分不计入已实现交易。
 95    """
 96    if fills is None or len(fills) == 0:
 97        return []
 98    df = fills.copy()
 99    df["timestamp"] = pd.to_datetime(df["timestamp"])
100    # 缺列兜底，兼容只给必要列的最简输入
101    for c in ("side", "symbol", "quantity", "exec_price", "total_fee"):
102        if c not in df.columns:
103            raise ValueError(f"成交明细缺少列: {c}")
104
105    pos_map: dict[pd.Timestamp, int] = {}
106    if equity_index is not None:
107        pos_map = {ts: i for i, ts in enumerate(pd.DatetimeIndex(equity_index))}
108
109    trips: list[RoundTrip] = []
110    for symbol, grp in df.groupby("symbol"):
111        grp = grp.sort_values("timestamp")
112        open_lots: list[dict] = []  # FIFO 队列：{qty, cost_per_share, entry_time}
113        for _, row in grp.iterrows():
114            side = str(row["side"]).upper()
115            qty = float(row["quantity"])
116            exec_price = float(row["exec_price"])
117            fee = float(row["total_fee"])
118            ts = pd.Timestamp(row["timestamp"])
119            if qty <= 0:
120                continue
121            if side == BUY:
122                cost_per_share = exec_price + fee / qty
123                open_lots.append(
124                    {"qty": qty, "cost": cost_per_share, "entry": ts, "entry_price": exec_price}
125                )
126            elif side == SELL:
127                proceeds_per_share = exec_price - fee / qty
128                remaining = qty
129                while remaining > 1e-12 and open_lots:
130                    lot = open_lots[0]
131                    matched = min(lot["qty"], remaining)
132                    pnl = (proceeds_per_share - lot["cost"]) * matched
133                    entry_cost = lot["entry_price"] * matched
134                    pnl_pct = pnl / entry_cost if entry_cost > 0 else 0.0
135                    hb = 1
136                    if pos_map:
137                        ei = pos_map.get(lot["entry"])
138                        xi = pos_map.get(ts)
139                        if ei is not None and xi is not None:
140                            hb = max(1, xi - ei)
141                    trips.append(
142                        RoundTrip(
143                            symbol=symbol,
144                            entry_time=lot["entry"],
145                            exit_time=ts,
146                            quantity=matched,
147                            entry_price=lot["entry_price"],
148                            exit_price=exec_price,
149                            pnl=float(pnl),
150                            pnl_pct=float(pnl_pct),
151                            win=pnl > 0,
152                            holding_bars=hb,
153                        )
154                    )
155                    lot["qty"] -= matched
156                    remaining -= matched
157                    if lot["qty"] <= 1e-12:
158                        open_lots.pop(0)
159    return trips

162# ---------------------------------------------------------------------------
163# 绩效分析器
164# ---------------------------------------------------------------------------
165class PerformanceAnalyzer:
166    """绩效与风险指标计算器（消费 A2 引擎输出的权益曲线 + 成交明细）。"""
167
168    def __init__(
169        self,
170        equity_curve: pd.DataFrame,
171        fills: pd.DataFrame | None = None,
172        periods_per_year: int = DEFAULT_PERIODS_PER_YEAR,
173        risk_free_rate: float = DEFAULT_RISK_FREE_RATE,
174    ):
175        self.equity_curve = equity_curve
176        self.equity = _equity_series(equity_curve)
177        self.returns = _returns_from_equity(self.equity)
178        self.fills = fills if fills is not None else pd.DataFrame()
179        self.periods_per_year = int(periods_per_year)
180        self.risk_free_rate = float(risk_free_rate)
181        self._trips: list[RoundTrip] | None = None
182
183    # ------------------------------ 收益 ------------------------------
184    def cumulative_return(self) -> float:
185        """累计收益率 = equity[-1]/equity[0] - 1。"""
186        if len(self.equity) < 2:
187            return 0.0
188        first = float(self.equity.iloc[0])
189        if first <= 0:
190            return 0.0
191        return float(self.equity.iloc[-1] / first - 1.0)
192
193    def annualized_return(self) -> float:
194        """年化收益率 = (1 + 累计收益) ** (periods_per_year / n_periods) - 1。"""
195        total = self.cumulative_return()
196        n = len(self.returns)
197        if n <= 0 or total <= -1.0:
198            return 0.0 if n <= 0 else -1.0
199        return float((1.0 + total) ** (self.periods_per_year / n) - 1.0)
200
201    def monthly_returns(self) -> pd.Series:
202        """月度收益：日收益率按自然月复利。"""
203        if self.returns.empty:
204            return pd.Series(dtype=float)
205        return self.returns.resample("ME").apply(lambda x: (1.0 + x).prod() - 1.0)
206
207    def yearly_returns(self) -> pd.Series:
208        """年度收益：日收益率按自然年复利。"""
209        if self.returns.empty:
210            return pd.Series(dtype=float)
211        return self.returns.resample("YE").apply(lambda x: (1.0 + x).prod() - 1.0)
212
213    # ------------------------------ 风险 ------------------------------
214    def annualized_volatility(self) -> float:
215        """年化波动率 = std(returns, ddof=1) * sqrt(periods_per_year)。"""
216        if len(self.returns) < 2:
217            return 0.0
218        return float(self.returns.std(ddof=1) * np.sqrt(self.periods_per_year))
219
220    def max_drawdown(self) -> float:
221        """最大回撤 = min(equity / cummax(equity) - 1)。"""
222        if len(self.equity) < 2:
223            return 0.0
224        dd = self.equity / self.equity.cummax() - 1.0
225        return float(dd.min())
226
227    def drawdown_series(self) -> pd.Series:
228        """回撤序列（每个时点相对历史最高权益的回撤）。"""
229        if len(self.equity) < 2:
230            return pd.Series(dtype=float)
231        return self.equity / self.equity.cummax() - 1.0
232
233    def max_drawdown_duration(self) -> int:
234        """最大回撤持续时间（bar 数，最长一段处于回撤状态的连续长度）。"""
235        dd = self.drawdown_series()
236        if dd.empty:
237            return 0
238        in_dd = dd < 0
239        if not in_dd.any():
240            return 0
241        groups = (in_dd != in_dd.shift()).cumsum()
242        durations = groups[in_dd].value_counts()
243        return int(durations.max()) if not durations.empty else 0
244
245    def sharpe_ratio(self) -> float:
246        """Sharpe = mean(excess)/std(excess, ddof=1) * sqrt(periods)。"""
247        excess = self.returns - self.risk_free_rate / self.periods_per_year
248        std = excess.std(ddof=1)
249        if len(excess) < 2 or std == 0 or np.isnan(std):
250            return 0.0
251        return float(excess.mean() / std * np.sqrt(self.periods_per_year))
252
253    def sortino_ratio(self) -> float:
254        """Sortino = mean(excess)/std(excess[excess<0], ddof=1) * sqrt(periods)。"""
255        excess = self.returns - self.risk_free_rate / self.periods_per_year
256        downside = excess[excess < 0]
257        if len(excess) < 2 or len(downside) < 1:
258            return 0.0
259        dstd = downside.std(ddof=1)
260        if dstd == 0 or np.isnan(dstd):
261            return 0.0
262        return float(excess.mean() / dstd * np.sqrt(self.periods_per_year))
263
264    def calmar_ratio(self) -> float:
265        """Calmar = 年化收益率 / |最大回撤|。"""
266        mdd = self.max_drawdown()
267        if mdd == 0:
268            return 0.0
269        return self.annualized_return() / abs(mdd)
270
271    def var_cvar(self, confidence: float = 0.95) -> tuple:
272        """VaR/CVaR：经验分位数口径。
273
274        var_idx = floor((1-confidence)*n)，clamp 到 [0, n-1]；CVaR = 收益 <= VaR 的均值。
275        """
276        r = self.returns.sort_values()
277        n = len(r)
278        if n == 0:
279            return 0.0, 0.0
280        var_idx = int((1.0 - confidence) * n)
281        var_idx = max(0, min(var_idx, n - 1))
282        var = float(r.iloc[var_idx])
283        tail = r.iloc[: var_idx + 1]
284        cvar = float(tail.mean()) if len(tail) > 0 else var
285        return var, cvar
286
287    # ------------------------------ 交易 ------------------------------
288    def round_trips(self) -> list[RoundTrip]:
289        """返回 FIFO 配对后的平仓交易列表（懒加载缓存）。"""
290        if self._trips is None:
291            self._trips = build_round_trips(self.fills, self.equity.index)
292        return self._trips
293
294    def trade_statistics(self) -> dict:
295        """交易统计：胜率、盈亏比、平均盈亏比、换手、持仓周期、成本占比。"""
296        trips = self.round_trips()
297        total = len(trips)
298        wins = [t for t in trips if t.win]
299        losses = [t for t in trips if not t.win]
300        n_wins, n_losses = len(wins), len(losses)
301        gross_profit = sum(t.pnl for t in wins)
302        gross_loss = abs(sum(t.pnl for t in losses))
303
304        win_rate = n_wins / total if total else 0.0
305        profit_factor = (
306            gross_profit / gross_loss
307            if gross_loss > 0
308            else (float("inf") if gross_profit > 0 else 0.0)
309        )
310        avg_win = gross_profit / n_wins if n_wins else 0.0
311        avg_loss = -gross_loss / n_losses if n_losses else 0.0
312        avg_win_loss_ratio = (
313            avg_win / abs(avg_loss) if avg_loss != 0 else (float("inf") if avg_win > 0 else 0.0)
314        )
315        avg_holding = float(np.mean([t.holding_bars for t in trips])) if total else 0.0
316
317        # 换手率 / 成本占比（基于成交额口径）
318        buy_value = sell_value = 0.0
319        total_fees = 0.0
320        total_traded_value = 0.0
321        if len(self.fills):
322            f = self.fills.copy()
323            f["exec_price"] = pd.to_numeric(f["exec_price"], errors="coerce")
324            f["quantity"] = pd.to_numeric(f["quantity"], errors="coerce")
325            f["total_fee"] = pd.to_numeric(f["total_fee"], errors="coerce")
326            value = f["exec_price"] * f["quantity"]
327            side = f["side"].astype(str).str.upper()
328            buy_value = float(value[side == BUY].sum())
329            sell_value = float(value[side == SELL].sum())
330            total_traded_value = buy_value + sell_value
331            total_fees = float(f["total_fee"].sum())
332        avg_equity = float(self.equity.mean()) if len(self.equity) else 0.0
333        turnover = total_traded_value / avg_equity if avg_equity > 0 else 0.0
334        cost_ratio = total_fees / total_traded_value if total_traded_value > 0 else 0.0
335
336        return {
337            "total_trades": total,
338            "winning_trades": n_wins,
339            "losing_trades": n_losses,
340            "win_rate": win_rate,
341            "profit_factor": _finite(profit_factor, 999.0),
342            "avg_win": avg_win,
343            "avg_loss": avg_loss,
344            "avg_win_loss_ratio": _finite(avg_win_loss_ratio, 999.0),
345            "turnover": turnover,
346            "avg_holding_bars": avg_holding,
347            "cost_ratio": cost_ratio,
348            "total_fees": total_fees,
349            "total_traded_value": total_traded_value,
350        }
351
352    # ------------------------------ 归因 ------------------------------
353    def attribute_by_symbol(self) -> pd.DataFrame:
354        """按标的归因：各 symbol 的平仓笔数、已实现盈亏、费用贡献。"""
355        trips = self.round_trips()
356        rows = []
357        for symbol in sorted({t.symbol for t in trips}):
358            st = [t for t in trips if t.symbol == symbol]
359            rows.append(
360                {
361                    "symbol": symbol,
362                    "trades": len(st),
363                    "realized_pnl": float(sum(t.pnl for t in st)),
364                    "win_rate": sum(1 for t in st if t.win) / len(st),
365                }
366            )
367        return pd.DataFrame(rows, columns=["symbol", "trades", "realized_pnl", "win_rate"])
368
369    # ------------------------------ 汇总 ------------------------------
370    def summary(self) -> dict:
371        """一次性输出全部指标（供报告与 C4 联动）。"""
372        var, cvar = self.var_cvar()
373        trade = self.trade_statistics()
374        start = self.equity.index.min() if len(self.equity) else pd.NaT
375        end = self.equity.index.max() if len(self.equity) else pd.NaT
376        return {
377            "start": start,
378            "end": end,
379            "n_periods": len(self.returns),
380            "total_return": self.cumulative_return(),
381            "annualized_return": self.annualized_return(),
382            "annualized_volatility": self.annualized_volatility(),
383            "sharpe_ratio": self.sharpe_ratio(),
384            "sortino_ratio": self.sortino_ratio(),
385            "calmar_ratio": self.calmar_ratio(),
386            "max_drawdown": self.max_drawdown(),
387            "max_drawdown_duration": self.max_drawdown_duration(),
388            "var_95": var,
389            "cvar_95": cvar,
390            **trade,
391        }
392
393    def to_dataframe(self) -> pd.DataFrame:
394        """把汇总指标转成两列 DataFrame（指标名/数值），供报告渲染。"""
395        return pd.DataFrame(list(self.summary().items()), columns=["指标", "数值"])

398def _finite(x: float, cap: float) -> float:
399    """把 inf/nan 归一为有限值（供盈亏比等无亏损时的兜底）。"""
400    if x is None or np.isnan(x):
401        return 0.0
402    if np.isinf(x):
403        return cap
404    return float(x)

408# 与 C4 联动：IS/OOS 两组权益序列对比（过拟合检测输入）
409# ---------------------------------------------------------------------------
410def compare_is_oos(
411    is_equity: pd.Series | pd.DataFrame,
412    oos_equity: pd.Series | pd.DataFrame,
413    periods_per_year: int = DEFAULT_PERIODS_PER_YEAR,
414    risk_free_rate: float = DEFAULT_RISK_FREE_RATE,
415) -> dict:
416    """对比样本内（IS）与样本外（OOS）两组权益序列。
417
418    接受 Series（权益值）或含 equity 列的 DataFrame。返回 IS/OOS 各自的

169    def __init__(
170        self,
171        equity_curve: pd.DataFrame,
172        fills: pd.DataFrame | None = None,
173        periods_per_year: int = DEFAULT_PERIODS_PER_YEAR,
174        risk_free_rate: float = DEFAULT_RISK_FREE_RATE,
175    ):
176        self.equity_curve = equity_curve
177        self.equity = _equity_series(equity_curve)
178        self.returns = _returns_from_equity(self.equity)
179        self.fills = fills if fills is not None else pd.DataFrame()
180        self.periods_per_year = int(periods_per_year)
181        self.risk_free_rate = float(risk_free_rate)
182        self._trips: list[RoundTrip] | None = None
183
184    # ------------------------------ 收益 ------------------------------
185    def cumulative_return(self) -> float:
186        """累计收益率 = equity[-1]/equity[0] - 1。"""
187        if len(self.equity) < 2:
188            return 0.0
189        first = float(self.equity.iloc[0])
190        if first <= 0:
191            return 0.0
192        return float(self.equity.iloc[-1] / first - 1.0)
193
194    def annualized_return(self) -> float:
195        """年化收益率 = (1 + 累计收益) ** (periods_per_year / n_periods) - 1。"""
196        total = self.cumulative_return()
197        n = len(self.returns)
198        if n <= 0 or total <= -1.0:
199            return 0.0 if n <= 0 else -1.0
200        return float((1.0 + total) ** (self.periods_per_year / n) - 1.0)
201
202    def monthly_returns(self) -> pd.Series:
203        """月度收益：日收益率按自然月复利。"""
204        if self.returns.empty:
205            return pd.Series(dtype=float)
206        return self.returns.resample("ME").apply(lambda x: (1.0 + x).prod() - 1.0)
207
208    def yearly_returns(self) -> pd.Series:
209        """年度收益：日收益率按自然年复利。"""
210        if self.returns.empty:
211            return pd.Series(dtype=float)
212        return self.returns.resample("YE").apply(lambda x: (1.0 + x).prod() - 1.0)
213
214    # ------------------------------ 风险 ------------------------------
215    def annualized_volatility(self) -> float:
216        """年化波动率 = std(returns, ddof=1) * sqrt(periods_per_year)。"""
217        if len(self.returns) < 2:
218            return 0.0
219        return float(self.returns.std(ddof=1) * np.sqrt(self.periods_per_year))
220
221    def max_drawdown(self) -> float:
222        """最大回撤 = min(equity / cummax(equity) - 1)。"""
223        if len(self.equity) < 2:
224            return 0.0
225        dd = self.equity / self.equity.cummax() - 1.0
226        return float(dd.min())
227
228    def drawdown_series(self) -> pd.Series:
229        """回撤序列（每个时点相对历史最高权益的回撤）。"""
230        if len(self.equity) < 2:
231            return pd.Series(dtype=float)
232        return self.equity / self.equity.cummax() - 1.0
233
234    def max_drawdown_duration(self) -> int:
235        """最大回撤持续时间（bar 数，最长一段处于回撤状态的连续长度）。"""
236        dd = self.drawdown_series()
237        if dd.empty:
238            return 0
239        in_dd = dd < 0
240        if not in_dd.any():
241            return 0
242        groups = (in_dd != in_dd.shift()).cumsum()
243        durations = groups[in_dd].value_counts()
244        return int(durations.max()) if not durations.empty else 0
245
246    def sharpe_ratio(self) -> float:
247        """Sharpe = mean(excess)/std(excess, ddof=1) * sqrt(periods)。"""
248        excess = self.returns - self.risk_free_rate / self.periods_per_year
249        std = excess.std(ddof=1)
250        if len(excess) < 2 or std == 0 or np.isnan(std):
251            return 0.0
252        return float(excess.mean() / std * np.sqrt(self.periods_per_year))
253
254    def sortino_ratio(self) -> float:
255        """Sortino = mean(excess)/std(excess[excess<0], ddof=1) * sqrt(periods)。"""
256        excess = self.returns - self.risk_free_rate / self.periods_per_year
257        downside = excess[excess < 0]
258        if len(excess) < 2 or len(downside) < 1:
259            return 0.0
260        dstd = downside.std(ddof=1)
261        if dstd == 0 or np.isnan(dstd):
262            return 0.0
263        return float(excess.mean() / dstd * np.sqrt(self.periods_per_year))
264
265    def calmar_ratio(self) -> float:
266        """Calmar = 年化收益率 / |最大回撤|。"""
267        mdd = self.max_drawdown()
268        if mdd == 0:
269            return 0.0
270        return self.annualized_return() / abs(mdd)
271
272    def var_cvar(self, confidence: float = 0.95) -> tuple:
273        """VaR/CVaR：经验分位数口径。
274
275        var_idx = floor((1-confidence)*n)，clamp 到 [0, n-1]；CVaR = 收益 <= VaR 的均值。
276        """
277        r = self.returns.sort_values()
278        n = len(r)
279        if n == 0:
280            return 0.0, 0.0
281        var_idx = int((1.0 - confidence) * n)
282        var_idx = max(0, min(var_idx, n - 1))
283        var = float(r.iloc[var_idx])
284        tail = r.iloc[: var_idx + 1]
285        cvar = float(tail.mean()) if len(tail) > 0 else var
286        return var, cvar

288    # ------------------------------ 交易 ------------------------------
289    def round_trips(self) -> list[RoundTrip]:
290        """返回 FIFO 配对后的平仓交易列表（懒加载缓存）。"""
291        if self._trips is None:
292            self._trips = build_round_trips(self.fills, self.equity.index)
293        return self._trips

295    def trade_statistics(self) -> dict:
296        """交易统计：胜率、盈亏比、平均盈亏比、换手、持仓周期、成本占比。"""
297        trips = self.round_trips()
298        total = len(trips)
299        wins = [t for t in trips if t.win]
300        losses = [t for t in trips if not t.win]
301        n_wins, n_losses = len(wins), len(losses)
302        gross_profit = sum(t.pnl for t in wins)
303        gross_loss = abs(sum(t.pnl for t in losses))
304
305        win_rate = n_wins / total if total else 0.0
306        profit_factor = (
307            gross_profit / gross_loss
308            if gross_loss > 0
309            else (float("inf") if gross_profit > 0 else 0.0)
310        )
311        avg_win = gross_profit / n_wins if n_wins else 0.0
312        avg_loss = -gross_loss / n_losses if n_losses else 0.0
313        avg_win_loss_ratio = (
314            avg_win / abs(avg_loss) if avg_loss != 0 else (float("inf") if avg_win > 0 else 0.0)
315        )
316        avg_holding = float(np.mean([t.holding_bars for t in trips])) if total else 0.0
317
318        # 换手率 / 成本占比（基于成交额口径）
319        buy_value = sell_value = 0.0
320        total_fees = 0.0
321        total_traded_value = 0.0
322        if len(self.fills):
323            f = self.fills.copy()
324            f["exec_price"] = pd.to_numeric(f["exec_price"], errors="coerce")
325            f["quantity"] = pd.to_numeric(f["quantity"], errors="coerce")
326            f["total_fee"] = pd.to_numeric(f["total_fee"], errors="coerce")
327            value = f["exec_price"] * f["quantity"]
328            side = f["side"].astype(str).str.upper()
329            buy_value = float(value[side == BUY].sum())
330            sell_value = float(value[side == SELL].sum())
331            total_traded_value = buy_value + sell_value
332            total_fees = float(f["total_fee"].sum())
333        avg_equity = float(self.equity.mean()) if len(self.equity) else 0.0
334        turnover = total_traded_value / avg_equity if avg_equity > 0 else 0.0
335        cost_ratio = total_fees / total_traded_value if total_traded_value > 0 else 0.0
336
337        return {
338            "total_trades": total,
339            "winning_trades": n_wins,
340            "losing_trades": n_losses,
341            "win_rate": win_rate,
342            "profit_factor": _finite(profit_factor, 999.0),
343            "avg_win": avg_win,
344            "avg_loss": avg_loss,
345            "avg_win_loss_ratio": _finite(avg_win_loss_ratio, 999.0),
346            "turnover": turnover,
347            "avg_holding_bars": avg_holding,
348            "cost_ratio": cost_ratio,
349            "total_fees": total_fees,
350            "total_traded_value": total_traded_value,
351        }
352
353    # ------------------------------ 归因 ------------------------------
354    def attribute_by_symbol(self) -> pd.DataFrame:
355        """按标的归因：各 symbol 的平仓笔数、已实现盈亏、费用贡献。"""
356        trips = self.round_trips()
357        rows = []
358        for symbol in sorted({t.symbol for t in trips}):
359            st = [t for t in trips if t.symbol == symbol]
360            rows.append(
361                {
362                    "symbol": symbol,
363                    "trades": len(st),
364                    "realized_pnl": float(sum(t.pnl for t in st)),
365                    "win_rate": sum(1 for t in st if t.win) / len(st),
366                }
367            )
368        return pd.DataFrame(rows, columns=["symbol", "trades", "realized_pnl", "win_rate"])

372        """一次性输出全部指标（供报告与 C4 联动）。"""
373        var, cvar = self.var_cvar()
374        trade = self.trade_statistics()
375        start = self.equity.index.min() if len(self.equity) else pd.NaT
376        end = self.equity.index.max() if len(self.equity) else pd.NaT
377        return {

379            "end": end,
380            "n_periods": len(self.returns),
381            "total_return": self.cumulative_return(),

383            "annualized_volatility": self.annualized_volatility(),
384            "sharpe_ratio": self.sharpe_ratio(),
385            "sortino_ratio": self.sortino_ratio(),
386            "calmar_ratio": self.calmar_ratio(),
387            "max_drawdown": self.max_drawdown(),
388            "max_drawdown_duration": self.max_drawdown_duration(),
389            "var_95": var,
390            "cvar_95": cvar,
391            **trade,
392        }

395        """把汇总指标转成两列 DataFrame（指标名/数值），供报告渲染。"""

def build_round_trips( fills: pandas.core.frame.DataFrame, equity_index: pandas.core.indexes.datetimes.DatetimeIndex | None = None) -> list[RoundTrip]: 
 88def build_round_trips(
 89    fills: pd.DataFrame,
 90    equity_index: pd.DatetimeIndex | None = None,
 91) -> list[RoundTrip]:
 92    """按 symbol 分组、FIFO 配对平仓，返回逐笔交易列表。
 93
 94    费用分摊：买入单位成本 = exec_price + 买入费/买入量；卖出单位收入 = exec_price - 卖出费/卖出量。
 95    未平仓（只有买入没有卖出）的部分不计入已实现交易。
 96    """
 97    if fills is None or len(fills) == 0:
 98        return []
 99    df = fills.copy()
100    df["timestamp"] = pd.to_datetime(df["timestamp"])
101    # 缺列兜底，兼容只给必要列的最简输入
102    for c in ("side", "symbol", "quantity", "exec_price", "total_fee"):
103        if c not in df.columns:
104            raise ValueError(f"成交明细缺少列: {c}")
105
106    pos_map: dict[pd.Timestamp, int] = {}
107    if equity_index is not None:
108        pos_map = {ts: i for i, ts in enumerate(pd.DatetimeIndex(equity_index))}
109
110    trips: list[RoundTrip] = []
111    for symbol, grp in df.groupby("symbol"):
112        grp = grp.sort_values("timestamp")
113        open_lots: list[dict] = []  # FIFO 队列：{qty, cost_per_share, entry_time}
114        for _, row in grp.iterrows():
115            side = str(row["side"]).upper()
116            qty = float(row["quantity"])
117            exec_price = float(row["exec_price"])
118            fee = float(row["total_fee"])
119            ts = pd.Timestamp(row["timestamp"])
120            if qty <= 0:
121                continue
122            if side == BUY:
123                cost_per_share = exec_price + fee / qty
124                open_lots.append(
125                    {"qty": qty, "cost": cost_per_share, "entry": ts, "entry_price": exec_price}
126                )
127            elif side == SELL:
128                proceeds_per_share = exec_price - fee / qty
129                remaining = qty
130                while remaining > 1e-12 and open_lots:
131                    lot = open_lots[0]
132                    matched = min(lot["qty"], remaining)
133                    pnl = (proceeds_per_share - lot["cost"]) * matched
134                    entry_cost = lot["entry_price"] * matched
135                    pnl_pct = pnl / entry_cost if entry_cost > 0 else 0.0
136                    hb = 1
137                    if pos_map:
138                        ei = pos_map.get(lot["entry"])
139                        xi = pos_map.get(ts)
140                        if ei is not None and xi is not None:
141                            hb = max(1, xi - ei)
142                    trips.append(
143                        RoundTrip(
144                            symbol=symbol,
145                            entry_time=lot["entry"],
146                            exit_time=ts,
147                            quantity=matched,
148                            entry_price=lot["entry_price"],
149                            exit_price=exec_price,
150                            pnl=float(pnl),
151                            pnl_pct=float(pnl_pct),
152                            win=pnl > 0,
153                            holding_bars=hb,
154                        )
155                    )
156                    lot["qty"] -= matched
157                    remaining -= matched
158                    if lot["qty"] <= 1e-12:
159                        open_lots.pop(0)
160    return trips
