102    @staticmethod
103    def max_drawdown_duration(equity: pd.Series) -> int:
104        peak = equity.expanding().max()
105        dd = equity / peak - 1
106        in_dd = dd < 0
107        if not in_dd.any():
108            return 0
109        groups = (in_dd != in_dd.shift()).cumsum()
110        durations = groups[in_dd].value_counts()
111        return int(durations.max()) if not durations.empty else 0

119    @staticmethod
120    def alpha_beta(returns: pd.Series, benchmark: pd.Series, rf: float = RISK_FREE_RATE) -> tuple:
121        aligned = pd.concat([returns, benchmark], axis=1).dropna()
122        if len(aligned) < 2:
123            return 0.0, 0.0
124        r = aligned.iloc[:, 0]
125        b = aligned.iloc[:, 1]
126        cov = np.cov(r, b)
127        if cov[1, 1] == 0:
128            return 0.0, 0.0
129        beta = cov[0, 1] / cov[1, 1]
130        alpha = (
131            r.mean()
132            - rf / MetricsCalculator.TRADING_DAYS
133            - beta * (b.mean() - rf / MetricsCalculator.TRADING_DAYS)
134        )
135        return float(alpha * MetricsCalculator.TRADING_DAYS), float(beta)

137    @staticmethod
138    def var_cvar(returns: pd.Series, confidence: float = 0.95) -> tuple:
139        sorted_ret = returns.dropna().sort_values()
140        if len(sorted_ret) == 0:
141            return 0.0, 0.0
142        var_idx = int((1 - confidence) * len(sorted_ret))
143        var_idx = max(0, var_idx)
144        var = float(sorted_ret.iloc[var_idx])
145        tail = sorted_ret.iloc[: var_idx + 1]
146        cvar = float(tail.mean()) if len(tail) > 0 else var
147        return var, cvar

149    @staticmethod
150    def trade_statistics(trades: pd.DataFrame) -> dict:
151        total = len(trades)
152        if total == 0:
153            return {
154                "total_trades": 0,
155                "winning_trades": 0,
156                "losing_trades": 0,
157                "win_rate": 0.0,
158                "profit_factor": 0.0,
159                "avg_win": 0.0,
160                "avg_loss": 0.0,
161            }
162
163        wins = trades[trades["win"]]
164        losses = trades[~trades["win"]]
165        total_wins = len(wins)
166        total_losses = len(losses)
167        avg_win = wins["pnl_pct"].mean() if total_wins > 0 else 0.0
168        avg_loss = losses["pnl_pct"].mean() if total_losses > 0 else 0.0
169
170        gross_profit = wins["pnl"].sum() if total_wins > 0 else 0.0
171        gross_loss = abs(losses["pnl"].sum()) if total_losses > 0 else 0.0
172        pf = gross_profit / gross_loss if gross_loss > 0 else float("inf")
173
174        return {
175            "total_trades": total,
176            "winning_trades": total_wins,
177            "losing_trades": total_losses,
178            "win_rate": total_wins / total if total > 0 else 0.0,
179            "profit_factor": pf if pf != float("inf") else 999.0,
180            "avg_win": avg_win,
181            "avg_loss": avg_loss,
182        }

168        avg_win = wins["pnl_pct"].mean() if total_wins > 0 else 0.0
169        avg_loss = losses["pnl_pct"].mean() if total_losses > 0 else 0.0
170
171        gross_profit = wins["pnl"].sum() if total_wins > 0 else 0.0
172        gross_loss = abs(losses["pnl"].sum()) if total_losses > 0 else 0.0
173        pf = gross_profit / gross_loss if gross_loss > 0 else float("inf")
174
175        return {
176            "total_trades": total,
177            "winning_trades": total_wins,
178            "losing_trades": total_losses,
179            "win_rate": total_wins / total if total > 0 else 0.0,
180            "profit_factor": pf if pf != float("inf") else 999.0,
181            "avg_win": avg_win,
182            "avg_loss": avg_loss,
183        }
